def trade_review_page():
    """
    Trade Review page: server-side render from today's log files.
    No required query params -> never 400 just because args are missing.
    """
    from flask import render_template
    import os, csv, datetime

    today = datetime.date.today().strftime("%Y-%m-%d")
    log_dir = os.path.join(os.getcwd(), "logs")

    buy_file  = os.path.join(log_dir, f"triggers_{today}.csv")
    sell_file = os.path.join(log_dir, f"sell_triggers_{today}.csv")

    def _read_rows(path, limit=5):
        rows = []
        if not os.path.exists(path):
            return rows
        with open(path, "r", encoding="utf-8", errors="ignore", newline="") as f:
            rdr = csv.DictReader(f)
            for r in rdr:
                rows.append(r)
        # newest first (logs are normally append; if yours are newest-last, reverse)
        # We'll assume append order => newest at end:
        rows = list(reversed(rows))
        return rows[:limit]

    buy_rows = _read_rows(buy_file, limit=5)
    sell_rows = _read_rows(sell_file, limit=5)

    return render_template(
        "trade_review.html",
        buy_rows=buy_rows,
        sell_rows=sell_rows,
        buy_file=buy_file if os.path.exists(buy_file) else None,
        sell_file=sell_file if os.path.exists(sell_file) else None,
    )

def trade_review_chart():
    """
    Trade Review card chart endpoint.
    Always return a valid Flask response (NOT a DataFrame).
    We reuse the existing /chart/<symbol> renderer and pass ts for the marker.
    """
    from flask import request, redirect, abort
    from urllib.parse import quote_plus

    symbol = (request.args.get("symbol") or "").strip().upper()
    ts = (request.args.get("buy_ts") or request.args.get("ts") or "").strip()

    if not symbol:
        abort(400, "Missing symbol")

    if ts:
        return redirect(f"/chart/{symbol}?ts={quote_plus(ts)}")

    return redirect(f"/chart/{symbol}")

def run_checkpoint():
    try:
        if not os.path.exists(CHECKPOINT_BAT):
            return {"ok": False, "error": f"Missing {CHECKPOINT_BAT}"}

        subprocess.Popen(
            ["cmd.exe", "/c", "start", "", CHECKPOINT_BAT],
            cwd=str(Path(CHECKPOINT_BAT).parent),
            creationflags=0x00000008,  # CREATE_NEW_CONSOLE
        )

        return {"ok": True, "launched": CHECKPOINT_BAT}
    except Exception as e:
        current_app.logger.exception("checkpoint error: %s", e)
        return {"ok": False, "error": str(e)}
